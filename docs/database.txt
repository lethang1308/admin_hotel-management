// MySQL >= 8.0.16; tiền VND dùng DECIMAL; datetime lưu UTC, ngày đêm phòng theo Asia/Ho_Chi_Minh.
// Đây là thiết kế đích, KHÔNG phải migration thay thế V1 của backend đang có.
// Các ràng buộc liên bảng và thuật toán khóa giao dịch được giải thích trong README.vi.md.

Project hotel_management {
  database_type: 'MySQL'
  Note: '''Quản lý MỘT khách sạn. Đầy đủ tài khoản/RBAC, phòng, giá động, đặt phòng nhiều phòng,
  xếp/đổi phòng, check-in/out, dịch vụ, hóa đơn, cọc/thanh toán/hoàn tiền, voucher,
  bảo trì/dọn phòng, thông báo, đánh giá, hỗ trợ và dữ liệu báo cáo.
  Khoảng lưu trú [check_in_date, check_out_date), ngày trả không tính đêm.
  UNIQUE room_night_claims(room_id, stay_date) + khóa room_type_inventory để chống bán trùng.
  Các bảng nghiệp vụ giữ lịch sử, không xóa cứng. Xem README.vi.md để triển khai transaction.'''
}

// ── 01. TÀI KHOẢN, PHÂN QUYỀN, HỒ SƠ ──────────────────────────────────────
Table users {
  id bigint [pk, increment]
  full_name varchar(120) [not null]
  email varchar(150) [not null, unique]
  phone varchar(20) [unique, note: 'Chuẩn hóa E.164; cho NULL nếu tài khoản cũ chưa có số. Đăng ký mới kiểm tra bắt buộc ở API.']
  password_hash varchar(255) [not null]
  avatar_url varchar(500)
  status varchar(30) [not null, default: 'ACTIVE', note: 'ACTIVE / LOCKED / DISABLED']
  email_verified_at datetime(6)
  last_login_at datetime(6)
  locked_until datetime(6)
  password_changed_at datetime(6)
  created_at datetime(6) [not null, default: `CURRENT_TIMESTAMP(6)`]
  updated_at datetime(6) [not null, default: `CURRENT_TIMESTAMP(6)`]
  indexes { status
    created_at
    full_name
  }
  checks { `status IN ('ACTIVE','LOCKED','DISABLED')` [name: 'ck_user_status'] }
  Note: 'Giữ tên bảng/cột xác thực gốc trong V1. Bổ sung cột mới bằng migration sau; không chạy lại CREATE lên database cũ.'
}

// ── 03. CHÍNH SÁCH, GIÁ ĐỘNG, VOUCHER ──────────────────────────────────────
Table deposit_policies {
  id bigint [pk, increment]
  code varchar(30) [not null]
  version int [not null, default: 1]
  name varchar(150) [not null]
  amount_type varchar(20) [not null, note: 'NONE / PERCENT / FIXED']
  amount_value decimal(15,2) [not null, default: 0]
  payment_due_minutes int [not null, default: 15]
  is_active boolean [not null, default: true]
  created_at datetime(6) [not null, default: `CURRENT_TIMESTAMP(6)`]
  indexes { (code, version) [unique] }
  checks { `amount_type IN ('NONE','PERCENT','FIXED') AND amount_value >= 0 AND payment_due_minutes > 0` [name: 'ck_deposit_values']
    `amount_type <> 'PERCENT' OR amount_value <= 100` [name: 'ck_deposit_percent']
    `amount_type <> 'NONE' OR amount_value = 0` [name: 'ck_deposit_none']
  }
  Note: 'Phiên bản đã được booking tham chiếu thì bất biến; thay chính sách bằng phiên bản mới.'
}

    `no_show_fee_type <> 'PERCENT' OR no_show_fee_value <= 100` [name: 'ck_no_show_percent']

  Note: 'Đổi chính sách tạo version mới. Mốc tính là giờ check-in dự kiến của từng booking_rooms.'

    `fee_type <> 'PERCENT' OR fee_value <= 100` [name: 'ck_cancel_rule_percent']

Table surcharge_policies {
  id bigint [pk, increment]
  code varchar(30) [not null]
  version int [not null, default: 1]
  name varchar(150) [not null]
  kind varchar(30) [not null, note: 'EARLY_CHECKIN / LATE_CHECKOUT / EXTRA_GUEST / EXTRA_BED']
  room_type_id bigint [ref: > room_types.id, note: 'NULL áp dụng mọi loại phòng']
  from_minutes int [note: 'Số phút nhận sớm/trả muộn, inclusive']
  to_minutes int [note: 'Exclusive; NULL không giới hạn']
  amount_type varchar(20) [not null, note: 'FIXED / PERCENT_NIGHT / PER_HOUR']
  amount_value decimal(15,2) [not null]
  priority int [not null, default: 0]
  is_active boolean [not null, default: true]
  indexes { (code, version) [unique]
    (kind, room_type_id, is_active)
  }
  checks { `kind IN ('EARLY_CHECKIN','LATE_CHECKOUT','EXTRA_GUEST','EXTRA_BED') AND amount_type IN ('FIXED','PERCENT_NIGHT','PER_HOUR') AND amount_value >= 0` [name: 'ck_surcharge_values']
    `from_minutes IS NULL OR from_minutes >= 0` [name: 'ck_surcharge_min']
    `to_minutes IS NULL OR (from_minutes IS NOT NULL AND to_minutes > from_minutes)` [name: 'ck_surcharge_range']
  }
  Note: 'Lưu phiên bản đã dùng vào booking_charges; không sửa lịch sử. PER_HOUR làm tròn lên số giờ tại service.'
}

Table rate_rules {
  id bigint [pk, increment]
  code varchar(40) [not null]
  version int [not null, default: 1]
  name varchar(150) [not null]
  kind varchar(30) [not null, note: 'NIGHT_OVERRIDE / LENGTH_DISCOUNT']
  start_date date [not null]
  end_date date [not null, note: 'Exclusive']
  weekday_mask int [not null, default: 127, note: 'Thứ 2=1, T3=2, ..., T7=32, CN=64; cuối tuần=96.']
  min_nights int [not null, default: 1]
  max_nights int
  value_type varchar(20) [not null, note: 'NIGHT_OVERRIDE dùng FIXED; LENGTH_DISCOUNT dùng PERCENT hoặc FIXED mỗi đêm.']
  value decimal(15,2) [not null]
  priority int [not null, default: 0]
  is_active boolean [not null, default: true]
  created_at datetime(6) [not null, default: `CURRENT_TIMESTAMP(6)`]
  indexes { (code, version) [unique]
    (kind, is_active, start_date, end_date)
  }
  checks { `end_date > start_date AND min_nights > 0 AND (max_nights IS NULL OR max_nights >= min_nights)` [name: 'ck_rate_dates_nights']
    `weekday_mask BETWEEN 1 AND 127 AND value >= 0` [name: 'ck_rate_values']
    `(kind = 'NIGHT_OVERRIDE' AND value_type = 'FIXED') OR (kind = 'LENGTH_DISCOUNT' AND value_type IN ('PERCENT','FIXED'))` [name: 'ck_rate_kind']
    `value_type <> 'PERCENT' OR value <= 100` [name: 'ck_rate_percent']
  }
